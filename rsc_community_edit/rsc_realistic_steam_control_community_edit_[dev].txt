[[----------------------------------------

@inputs SlipMPH

DPS = WheelRPM * 6 #Degrees per second
WheelAng = (WheelAng + (DPS*Interval)) % 360

[[-----------------------------------------
'Interval' and 'Diameter' should be substituted for values relevant to your wheelbase.
'Interval' is in ms, so if your interval rate is 50, 'Interval' should equal 0.05

SETUP INSTRUCTIONS:
0 - Configure chip.
1 - Place the e2 on the rear of a steam locomotive's baseplate and leave it welded.
2 - Create an ADVANCED Entity Marker and link it to all the locomotive's DRIVER wheels if if uses a PROP base.

Use the blower to increase fire temp at a stall. Fire transfers more heat to the boiler the hotter it gets.

    #This function "publishes" the in-progress data to the semi-permanent data at the end of each train scan.
    function void publishTrainData(){
        Train = TrainInProgress
        if(CarsAheadInProgress > 0){
            TrainStringInProgress = "("+CarsAheadInProgress+"c) " + TrainStringInProgress
        }
        if(CarsBehindInProgress > 0){
            TrainStringInProgress = TrainStringInProgress + " ("+CarsBehindInProgress+"c)"
        }
        CarsAhead = CarsAheadInProgress
        CarsBehind = CarsBehindInProgress
        
        TrainString = TrainStringInProgress
        
        UnitsAhead = UnitsAheadInProgress
        UnitsBehind = UnitsBehindInProgress
        TotalUnits = UnitsAhead+UnitsBehind+1
        TotalCars = CarsAhead+CarsBehind
        
        TrainEnts = KnownEnts
        
        #Calculate Brake Length, the max number of effective cars you can get away from a locomotive.
        #also trailing tons
        #and total horsepower why not
        local BLIP = 0
        local BLMAX = 0
        local FoundLoco = 0
        local TrailingMassInProgress = 0
        local TrainHorsepowerInProgress = 0
        local TotalBrakeWorth = 0
        for(N = -UnitsAhead, UnitsBehind){
            local Unit = Train[N,table]
            if(Unit["isloco",number]){ #It's a locomotive
                if(FoundLoco){ #There was a locomotive ahead of this one
                    if(BLIP/2 > BLMAX){
                        BLMAX = BLIP/2
                    }
                }else{ #This is the first locomotive found in the consist
                    FoundLoco = 1
                    if(BLIP > BLMAX){
                        BLMAX = BLIP
                    }
                }
                BLIP = 0
                TrainHorsepowerInProgress += Unit["horsepower",number]
            }else{ #It's a car
                local BW = Unit["brakeworth",number]
                BLIP += BW
                TrailingMassInProgress += Unit["mass",number]
                TotalBrakeWorth += BW
            }
        }
        if(BLIP > BLMAX){BLMAX = BLIP}
        BrakeLength = BLMAX
        TrailingTons = TrailingMassInProgress*2/907.2 #kg to tons, then RLC mass scale
        TrainHorsePower = TrainHorsepowerInProgress
        
        MaxTrainBrakeForce = BrakeMul*TotalBrakeWorth
        
        HPT = round(TrainHorsePower/max(TrailingTons,1),2)
        IHPT = round(IHP/max(TrailingTons,1),2)
        
        #print(TrainString)
        ScanTimeStamp = curtime()
        ScanPhase = 4 #Go into Cooldown
    }

                            local KeepExploring = 1

                                #print(Sample)
                                
                                #Check/add Mass
                                if(CheckM){
                                    local M = Sample:mass()
                                    if(M > HeaviestMass){
                                        HeaviestMass = M
                                        HeaviestEnt = Sample
                                    }
                                    UnitMass += M
                                    
                                    if(M > MassThreshold){ #Prop is heavy enough to be considered part of the rolling stock
                                        PropCount++
                                        Props:pushEntity(Sample)
                                    }
                                    
                                    #Check if it's a locomotive
                                    if(Sample:type()=="gmod_wire_expression2"){ #Entity is an E2
                                        if(Sample["rlcpt2_isloco",number]){ #Entity is an RLC PT2 control chip
                                            IsLoco = 1
                                            UnitHP = Sample["rlcpt2_horsepower",number]
                                            #print(Sample["rlcpt2_horsepower",number])
                                        }
                                    }
                                    CheckM = 0
                                }
                                
                                #Find welded entities and add them to the queue
                                if(CheckW){
                                    local W_Ent = Sample:isConstrainedTo("weld",Welds+1)
                                    if(W_Ent){ #Found an entity welded to the Sample
                                        Welds++
                                        if(!KnownEnts[W_Ent:id(),number]){ #The entity is new
                                            UnitEnts:pushEntity(W_Ent)
                                            KnownEnts[W_Ent:id(),number] = 1
                                        }
                                    }else{
                                        CheckW = 0
                                    }
                                }
                                
                                #Find axised entities and add them to the queue
                                if(CheckA){
                                    local A_Ent = Sample:isConstrainedTo("axis",Axes+1)
                                    if(A_Ent){ #Found an entity axised to the Sample
                                        Axes++
                                        if(!KnownEnts[A_Ent:id(),number]){ #The entity is new
                                            UnitEnts:pushEntity(A_Ent)
                                            KnownEnts[A_Ent:id(),number] = 1
                                        }
                                    }else{
                                        CheckA = 0
                                    }
                                }
                                
                                #Find roped entities and mark it for the next car

                                    local R_Ent = Sample:isConstrainedTo("rope",Ropes+1)
                                    if(R_Ent){ #Found an entity axised to the Sample
                                        Ropes++
                                        if(!KnownEnts[R_Ent:id(),number]){ #The entity is new
                                            RCount++
                                            RopeEnts:pushEntity(R_Ent)
                                            KnownEnts[R_Ent:id(),number] = 1
                                        }
                                    }else{
                                        CheckR = 0
                                    }
                                }
                                
                                if(!(CheckW | CheckA | CheckR)){ #No new constrained entities, move on to the next one

                                        if(ScanPhase==1){ #Origin Unit

    function number versionCheck(Version:string,Expected:string){
        Conflict = 0
        #print("Comparing " + Version + " to " + Expected + ".")
        
        local VExp = Version:explode(".")
        local VMaj = VExp[1,string]:toNumber()
        local VMin = VExp[2,string]:toNumber()
        local VRev = VExp[3,string]:toNumber()
        
        
        local EExp = Expected:explode(".")
        local EMaj = EExp[1,string]:toNumber()
        local EMin = EExp[2,string]:toNumber()
        local ERev = EExp[3,string]:toNumber()
    
    
        if((VMaj > EMaj) | (VMin > EMin) | (VRev > ERev)){return -1} #Dev build
        if((VMaj < EMaj) | (VMin < EMin) | (VRev < ERev)){return 1}  #Invalid version
        
        
        return 0
    }
    
    if(versionCheck(ConfigVer,MinConfigVer) == 1){printColor(vec(255,100,0),"[RSC] Config version outdated. Proceed with caution and consider updating.")}
    if(versionCheck(ControlVer,MinControlVer) == 1){printColor(vec(255,100,0),"[RSC] Controls version outdated. Proceed with caution and consider updating.")}

        #owner():soundPlay("v_err",0,"garrysmod/content_downloaded.wav")
    }
    
    if(Conflict == 0){#Up to Date
        printColor(vec(36,255,36),"[RSC] RSC is up to date!")
    }
    
    if(Conflict == -1){#Dev-build
        printColor(vec(255,255,0),"[RSC] RSC version ahead of source.")
    }
}

timer("fireman",FiremanDt,0,function(){
    
    #[
          /  M  \
         /-------\
        / D  |  V \
    ]#
    
    if(Fireman & Active){
    
        #FIREBOX ------------------------------------------------------------------------------
        
        if(Stoker){
            Stoker = clamp(Stoker,0,1)
            local CoalToAdd = StokerRate * FiremanDt * Stoker #Mass of stoker to add
            local StokerVolume = StokerRate / FuelDensity
            GrateMass = GrateMass + CoalToAdd #Calculate coal mass (lb)
            FireTemp = ((StokerVolume * 1500) + (FuelVolume * FireTemp))/(StokerVolume + FuelVolume)
        }
        
        if(BoilerSoundMin > WaterHeight){CrownSheetHealth -= 1*FiremanDt}#-1 HP/S
        
        GrateEnergy = GrateMass * FuelEnergy #Calculate BTU of grate
        #1 BTU to increase 1lb of water's temp by 1F
        #970 BTU/lb at boiling point to generate steam.
        
        local TheoreticalTemp = GrateEnergy/(AirHeatCapacity*FireboxMass)
        local RequiredAir = GrateMass * AirFuelRatio
        
        InjectA = clamp(InjectA,0,1)
        InjectB = clamp(InjectB,0,1)
        Inject = clamp(InjectA + InjectB,0,2)
        Blower = clamp(Blower,0,1)
        Damper = clamp(Damper,0,1)
        #local DraftFactor = max(Blower * 0.15,clamp((WheelRPM/750)*0.6,0,0.6)) + (Damper ? clamp((MPH/60)*0.4,0,0.4) : 0) + 0.1
        local FPM = MPH * 88 #FEET??!!?? PER MINUTE??>>?
        local BlowerCFM = lerp(0,20000,Blower) #3000-20000 CFM
        local DamperCFM = DamperArea * FPM * Damper #Air mass flow through damper
        local EPM = (WheelRPM*(2*CylinderCount)) #Exhausts Per Minute
        local DraftCFM = EPM*(SmokeBoxVolume*0.8) * sqrt(ThrottleN) #Pull 80% of the smokebox's volume per exhaust * throttle (I think the math works out to that from EPM????)
        local Choke = (1-clamp((GrateHeight-5)/3,0,1)) #Too much coal, starving the fire
        local AirMassflow = (((BlowerCFM + DamperCFM + DraftCFM + 0.5)*AirDensity)/60)*Choke*FiremanDt #CFS/t (I hope)
        
        #We burn a little coal
        if(GrateMass > 0){
            local BurnEff = AirMassflow/RequiredAir
            
            local ArrEq = 1/(1+2^(12-(0.0075*FireTemp)))
                    #print(_HUD_PRINTCENTER,"" + ArrEq)
                        #lerp(0,1,(FireTemp-900)/2700)
                        
                        #0.0025/lb/coal/second/grate sqin #*0.0025*GrateL*GrateW
            BurnRate = (AirMassflow/AirFuelRatio)*(1+ArrEq)*FiremanDt #(ArrEq) #Burned this tick
            
            GrateMass -= (BurnRate)
            #if(GrateMass < 0){GrateMass = 0}
        }else{
            GrateMass = 0
            BurnRate = 0
        }
        
        FuelVolume = GrateMass/FuelDensity #Calculate coal volume
        
        #Fire thermal output to temperature
        FireboxOutput = BurnRate*(FuelEnergy) #BTU
        FireTemp += (FireboxOutput/(AirHeatCapacity*FireboxMass)) #Increase temp
        FireTemp = clamp(FireTemp,1000,3000) #Clamp it
        
        #Fire color
        local FireFactor = lerp(0,1,(FireTemp-900)/(3000-900)) #Color factor
        Firebox = vec(255,255*FireFactor,240*FireFactor^2)
        BurnQuality = sqrt(FireFactor)
        
        #holoColor(FireHolo,Firebox)
        
        #Determine fuel height on grate
        local GrateLFt = GrateL/12 #Convert to feet
        local GrateWFt = GrateW/12
        GrateHeight = (FuelVolume/(GrateLFt*GrateWFt))*12 #Convert back to inches
        
        #BOILER ------------------------------------------------------------------------------
        
        local PSIA = BoilerPSI + 14.7
        local BoilingPoint = 212+(12*sqrt(PSIA-14.7)) #Potbob boiling in a sponge of shit
        
        #Transfer heat from firebox to water
        #TempDiff = FireTemp - WaterTemp
        HeatTransfered = FireTemp * lerp(0,1,(FireTemp-900)/(1500-900)) #Basic equation that transfers more heat at a higher temperature
        WaterTemp += FireboxOutput/WaterMass #Water is 1:1 with BTU
        FireTemp -= FireboxOutput/WaterMass #Decrease fire temp
        WaterEnergy += FireboxOutput #Add thermal energy lost from fire to water, 1:1 for now
        local LatentHeat = lerp(970.1,714.36,BoilerPSI/680)#42*sqrt(-WaterTemp + 740)#970.1 at 212 F
        WaterEnthalpy = WaterEnergy
        
        #-k*(T-Ta)
        #T - Water temp
        #Ta - atmos temp
        #-k transfer constant - steel is ~ 0.001/15
        
        #Boiler
        
        
        if(Safeties){#Pop saftey valves
           SteamMass -= 15*Dt #Exhaust 15lb of steam per second
        }
        
        if(WaterTemp > BoilingPoint){#Let's boil some water
            #EvapRate = (FireboxOutput/LatentHeat) #Lb/dT of steam/s, latent heat
            
            local Diff = WaterTemp - BoilingPoint #Temperature difference
            local EnergyDiff = WaterMass*Diff #BTU of temperature difference
            EvapRate = (EnergyDiff/LatentHeat)*Dt #Evaporate the amount of difference, should work for normal boiling scenarios and flashing?
            #print(_HUD_PRINTCENTER,
            #"Temp diff: " + Diff +
            #"\nEnergy diff BTU: " + round(EnergyDiff,1))
            
            #local EnergyRelease = Diff
            WaterTemp -= Diff*Dt #Decrease temperature
            WaterEnergy -= EnergyDiff
            SteamEnergy += EnergyDiff
            
        }else{#nvm
            EvapRate = 0
        }
        
        WaterTemp = max(AmbientTemp,WaterTemp)#Minimum ambient
        
        #Evaporate water mass
        WaterMass -= (EvapRate)
        WaterVolume = WaterMass/WaterDensity
        
        #Blowdown
        local Blowdown = clamp(BlowdownA + BlowdownB,0,2)
        if(Blowdown){
            local BlowdownVolume = ((5/60)*WaterDensity)*FiremanDt*Blowdown
            WaterVolume -= BlowdownVolume
        }
        
        if(Inject & FillFactor < 0.95){#Inject water and alter temperature, don't fill if boiler is over 95% of fill
            #local Orifice = _PI * 1.5^2
            local InjectVolume = ((10/60)*WaterDensity)*FiremanDt*Inject #In 1 tick, 10 CFM at full blast
            WaterVolume += InjectVolume #10 CFM of water #((BoilerCap/7.48) * FiremanDt * 0.005)*Inject
            
            #Equation for thermal difference
            #((Vol1*F1) + (Vol2*F2))/(Vol1 + Vol2) = Final temperature
            
            WaterTemp = ((InjectVolume * (FeedwaterHeater ? WaterTemp*0.9 : AmbientTemp)) + (WaterVolume * WaterTemp))/(InjectVolume + WaterVolume)
        }
        
        #Calculate water mass and clamp volume
        WaterVolume = clamp(WaterVolume,0,(BoilerCap/7.48))
        WaterMass = WaterDensity * WaterVolume
        
        #Determine fill height and sight glass
        FillFactor = WaterVolume/BoilerVolume 
        WaterHeight = lerp(0,BoilerD,FillFactor)
        SightGlass = clamp(((WaterHeight-BoilerSoundMin)/SightGlassHeight),0,1)*100
        
        
        #STEAM --------------------------------------------------------------------------------
        
        #Add evap rate to steam mass
        SteamMass += EvapRate
        SteamMass = max(0,SteamMass) #Clamp at 0
        BoilerUllage = BoilerVolume*(1-FillFactor)#Empty space of boiler
        
        SteamSpecificVol = BoilerUllage/SteamMass #ft^3/lb
        local GPSI = ((1/(SteamSpecificVol-0.05))*440)-16.5 #Rough curve from https://www.engineeringtoolbox.com/saturated-steam-properties-d_273.html
        
        BoilerPSI = max(0,GPSI)#clamp(GPSI,0,BoilerMaxPSI)
        
        
        #SteamEnthalpy
        
        #SteamTemp = 0
        
        #SOUND --------------------------------------------------------------------------------
        
        local InjectS = sign(Inject)
        if(changed(InjectS) & InjectS){
            Pod:soundPlay("injector",0,InjectorOn)
        }elseif(changed(sign(Inject)) & !InjectS){
            soundStop("injector",1)
            #SndHolo5:soundPlay("injector",0,InjectorOff)
        }
        soundVolume("injector",sqrt(Inject))
        
        local BlowerS = sign(Blower)
        if(changed(BlowerS) & BlowerS){
            Pod:soundPlay("blower",0,BlowerOn)
        }elseif(changed(BlowerS) & !BlowerS){
            #SndHolo5:soundPlay("blower",0,BlowerOff)
            soundStop("blower",2)
        }
        soundVolume("blower",sqrt(Blower))
        
        local StokerS = sign(Stoker)
        if(changed(StokerS) & StokerS){
            Pod:soundPlay("stoker",0,StokerLoop)
        }elseif(changed(StokerS) & !StokerS){
            #Base:soundPlay("stoker",0,StokerOff)
            soundStop("stoker",1)
        }
        soundVolume("stoker",sqrt(Stoker))
        
        local BlowdownS = sign(Blowdown)
        if(changed(BlowdownS) & BlowdownS){
            Pod:soundPlay("blowdown",0,BlowdownOn)
        }elseif(changed(sign(BlowdownS)) & !BlowdownS){
            soundStop("blowdown",1)
        }
        
        #print(_HUD_PRINTCENTER,"InjectS " + InjectS + "\nBlowerS " + BlowerS + "\nBlower " + Blower)
        
        #print(_HUD_PRINTCENTER, "" + sign(Blower))
        #WaterTemp -= ()
        #[
        print(_HUD_PRINTCENTER,
            "Coal LBS: " + round(GrateMass,1) +
            "\nCoal Volume: " + round(FuelVolume,2) +
            "\nCoal Height: " + round(GrateHeight,1) +
            "\nGrate Energy: " + round(GrateEnergy) +
            #"\nTargetTemp: " + round(TheoreticalTemp) +
            #"\nRequired Air: " + round(RequiredAir,1) +
            "\nAirmass Lb/dT: " + round(AirMassflow,2) +
            "\nBurn Rate Lb/dT:" + round(BurnRate,3) +
            #"\nFirebox Output:" + round(FireboxOutput) +
            "\nFirebox Temp:" + round(FireTemp,1) +
            "\n\nWater Temp:" + round(WaterTemp,2) +
            #"\nCrown Sheet Health:" + round(CrownSheetHealth) +
            
            #"\nWater Mass: " + round(WaterMass) +
            #"\nWater Gal: " + round(WaterVolume*7.48) +
            #"\nFill Factor: " + round(FillFactor,2) +
            #"\nBTUs required to boil:" + round((212-WaterTemp)*WaterMass) +
            "\nEvap Rate LB/s: " + round(EvapRate,1) +

        )
        ]#
        BOILER_DATA = table(
