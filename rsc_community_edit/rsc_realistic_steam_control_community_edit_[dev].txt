SETUP FOR HOLO WHEELBASE:
Insert and initalize these values inside your wheel code:
[[----------------------------------------

@inputs MPH SlipMPH
@persist DPS WheelAng

WheelRPM = ((MPH+SlipMPH)*336)/Diameter
DPS = WheelRPM * 6 #Degrees per second
WheelAng = (WheelAng + (DPS*Interval)) % 360

[[-----------------------------------------
'Interval' and 'Diameter' should be substituted for values relevant to your wheelbase.
'Interval' is in ms, so if your interval rate is 50, 'Interval' should equal 0.05

SETUP INSTRUCTIONS:
0 - Configure chip.
1 - Place the e2 on the rear of a steam locomotive's baseplate and leave it welded.
2 - Create an ADVANCED Entity Marker and link it to all the locomotive's DRIVER wheels if if uses a PROP base.
    2.1 - If the locomotive uses a HOLO base, mark all sliders/axles to the entity marker INCLUDING leading/trailing trucks.
3 - Wire Drivers[ARRAY] to the entity marker if it uses a PROP base, otherwise wire Trucks[ARRAY] to the entity marker.
4 - Wire Pod:entity to the locomotive's pod controller Entity:entity output.
5 - Wire Ignite to either a pod controller key or a NON-TOGGLED button.
6 - If manual fireman mode is desired: wire Damper, AddCoal, and AddWater to either pod controller keys or wire buttons (Can be toggled or non-toggled).
7 - Set up screens, gauges, indicators, heiroglyphs, whatever so you can see relevant information like speed, throttle setting, water levels, etc.

The rest of the steps are optional but neat.
8 - Put a wire light on the firebox and wire it to Firebox[VECTOR]. It'll glow in accordance with the temperature level!
9 - Wire the Base1 through Base4 inputs to the 4 cylinder cock particle emitters on the pistons, and Wire C1L through C2R to their corresponding outputs on the holobase.
10- Wire up the smoke/whistle particles, remember to wire "interval" on the wire emitters to the On outputs alongside everything else you wire


One last little tip: the "CONSTANT" output is a constant value of 1. Very Handy.

FIREMAN'S GUIDE:
Keep Coal and Water around 100. If you add too much of either, temperature and steam generation gets nerfed.
Keep Temperature and Steam as high as you can. Max is 200.
If your temperature is sinking, use the Damper. It will boost your temperature greatly but will eat a lot more coal.

DRIVER'S GUIDE:
The Reverser increments by 1/4ths from -1 to +1. More extreme reverser values (closer to 1) will give you more starting force but a lower top speed.
As you approach the top of your "power band," you will stop accelerating until you lower the reverser.
Reverser values of 0.125 or -0.125 give you maximum horsepower (and by extension top speed) but you won't get any starting force.

Happy Steaming!

    #This function "publishes" the in-progress data to the semi-permanent data at the end of each train scan.
    function void publishTrainData(){
        Train = TrainInProgress
        if(CarsAheadInProgress > 0){
            TrainStringInProgress = "("+CarsAheadInProgress+"c) " + TrainStringInProgress
        }
        if(CarsBehindInProgress > 0){
            TrainStringInProgress = TrainStringInProgress + " ("+CarsBehindInProgress+"c)"
        }
        CarsAhead = CarsAheadInProgress
        CarsBehind = CarsBehindInProgress
        
        TrainString = TrainStringInProgress
        
        UnitsAhead = UnitsAheadInProgress
        UnitsBehind = UnitsBehindInProgress
        TotalUnits = UnitsAhead+UnitsBehind+1
        TotalCars = CarsAhead+CarsBehind
        
        TrainEnts = KnownEnts
        
        #Calculate Brake Length, the max number of effective cars you can get away from a locomotive.
        #also trailing tons
        #and total horsepower why not
        local BLIP = 0
        local BLMAX = 0
        local FoundLoco = 0
        local TrailingMassInProgress = 0
        local TrainHorsepowerInProgress = 0
        local TotalBrakeWorth = 0
        for(N = -UnitsAhead, UnitsBehind){
            local Unit = Train[N,table]
            if(Unit["isloco",number]){ #It's a locomotive
                if(FoundLoco){ #There was a locomotive ahead of this one
                    if(BLIP/2 > BLMAX){
                        BLMAX = BLIP/2
                    }
                }else{ #This is the first locomotive found in the consist
                    FoundLoco = 1
                    if(BLIP > BLMAX){
                        BLMAX = BLIP
                    }
                }
                BLIP = 0
                TrainHorsepowerInProgress += Unit["horsepower",number]
            }else{ #It's a car
                local BW = Unit["brakeworth",number]
                BLIP += BW
                TrailingMassInProgress += Unit["mass",number]
                TotalBrakeWorth += BW
            }
        }
        if(BLIP > BLMAX){BLMAX = BLIP}
        BrakeLength = BLMAX
        TrailingTons = TrailingMassInProgress*2/907.2 #kg to tons, then RLC mass scale
        TrainHorsePower = TrainHorsepowerInProgress
        
        MaxTrainBrakeForce = BrakeMul*TotalBrakeWorth
        
        HPT = round(TrainHorsePower/max(TrailingTons,1),2)
        IHPT = round(IHP/max(TrailingTons,1),2)
        
        #print(TrainString)
        ScanTimeStamp = curtime()
        ScanPhase = 4 #Go into Cooldown
    }
    
    function void carScan(){
        local W = Base
        if(MPH < 10){ #Train is stopped or moving slowly
                #if(Tick){
                    if(ScanPhase==0){ #Init to be done before scanning train
                        TrainInProgress = table()
                        TrainStringInProgress = ""
                        KnownEnts = table()
                        KnownEnts[W:id(),number] = 1 #Store the first entity
                        NextStartingEnt_F = noentity()
                        NextStartingEnt_R = noentity()
                        NextStartingEnt = W
                        
                        UnitIndex = 0
                        CarsAheadInProgress = CarsBehindInProgress = UnitsAheadInProgress = UnitsBehindInProgress = 0
                        
                        ScanPhase = 1 #Begin scanning the origin unit (unit E2 is attached to)
                    }elseif((ScanPhase>0) & (ScanPhase<4)){ #Scan!
                        if(NextStartingEnt){ #There is rolling stock to scan
                            #Init to be done before scanning a unit
                            local UnitMass = 0
                            local HeaviestMass = 0
                            local HeaviestEnt = W
                            local IsLoco = 0
                            local PropCount = 0
                            local Props = array()
                            local UnitHP = 0
                            
                            local Welds = 0
                            local Axes = 0
                            local Ropes = 0
                            
                            local CheckW = 1
                            local CheckA = 1
                            local CheckR = 1
                            local CheckM = 1
                            
                            UnitEnts:clear()
                            UnitEnts[1,entity] = NextStartingEnt #Store the first entity
                            local Sample = NextStartingEnt
                            
                            local RopeEnts = array()
                            local RCount = 0
                            NextStartingEnt = noentity()
                            ExpIndex = 1
                            
                            local KeepExploring = 1
                            while(KeepExploring){
                                #print(Sample)
                                
                                #Check/add Mass
                                if(CheckM){
                                    local M = Sample:mass()
                                    if(M > HeaviestMass){
                                        HeaviestMass = M
                                        HeaviestEnt = Sample
                                    }
                                    UnitMass += M
                                    
                                    if(M > MassThreshold){ #Prop is heavy enough to be considered part of the rolling stock
                                        PropCount++
                                        Props:pushEntity(Sample)
                                    }
                                    
                                    #Check if it's a locomotive
                                    if(Sample:type()=="gmod_wire_expression2"){ #Entity is an E2
                                        if(Sample["rlcpt2_isloco",number]){ #Entity is an RLC PT2 control chip
                                            IsLoco = 1
                                            UnitHP = Sample["rlcpt2_horsepower",number]
                                            #print(Sample["rlcpt2_horsepower",number])
                                        }
                                    }
                                    CheckM = 0
                                }
                                
                                #Find welded entities and add them to the queue
                                if(CheckW){
                                    local W_Ent = Sample:isConstrainedTo("weld",Welds+1)
                                    if(W_Ent){ #Found an entity welded to the Sample
                                        Welds++
                                        if(!KnownEnts[W_Ent:id(),number]){ #The entity is new
                                            UnitEnts:pushEntity(W_Ent)
                                            KnownEnts[W_Ent:id(),number] = 1
                                        }
                                    }else{
                                        CheckW = 0
                                    }
                                }
                                
                                #Find axised entities and add them to the queue
                                if(CheckA){
                                    local A_Ent = Sample:isConstrainedTo("axis",Axes+1)
                                    if(A_Ent){ #Found an entity axised to the Sample
                                        Axes++
                                        if(!KnownEnts[A_Ent:id(),number]){ #The entity is new
                                            UnitEnts:pushEntity(A_Ent)
                                            KnownEnts[A_Ent:id(),number] = 1
                                        }
                                    }else{
                                        CheckA = 0
                                    }
                                }
                                
                                #Find roped entities and mark it for the next car
                                if(CheckR){
                                    local R_Ent = Sample:isConstrainedTo("rope",Ropes+1)
                                    if(R_Ent){ #Found an entity axised to the Sample
                                        Ropes++
                                        if(!KnownEnts[R_Ent:id(),number]){ #The entity is new
                                            RCount++
                                            RopeEnts:pushEntity(R_Ent)
                                            KnownEnts[R_Ent:id(),number] = 1
                                        }
                                    }else{
                                        CheckR = 0
                                    }
                                }
                                
                                if(!(CheckW | CheckA | CheckR)){ #No new constrained entities, move on to the next one
                                    ExpIndex++
                                    Sample = UnitEnts[ExpIndex,entity]
                                    CheckW = CheckA = CheckR = CheckM = 1
                                    Welds = Axes = Ropes = 0
                                    if(!Sample){ #We've finished the unit!
                                        KeepExploring = 0
                                        
                                        #Store unit data into Train table
                                        local UnitTable = table()
                                        UnitTable["carbody",entity] = HeaviestEnt
                                        UnitTable["propcount",number] = PropCount
                                        UnitTable["mass",number] = UnitMass
                                        UnitTable["isloco",number] = IsLoco
                                        UnitTable["props",array] = Props
                                        UnitTable["brakeworth",number] = max(1, floor(PropCount/3))
                                        UnitTable["horsepower",number] = UnitHP

    function number versionCheck(Version:string,Expected:string){
        Conflict = 0
        #print("Comparing " + Version + " to " + Expected + ".")
        
        local VExp = Version:explode(".")
        local VMaj = VExp[1,string]:toNumber()
        local VMin = VExp[2,string]:toNumber()
        local VRev = VExp[3,string]:toNumber()
        
        
        local EExp = Expected:explode(".")
        local EMaj = EExp[1,string]:toNumber()
        local EMin = EExp[2,string]:toNumber()
        local ERev = EExp[3,string]:toNumber()
    
    
        if((VMaj > EMaj) | (VMin > EMin) | (VRev > ERev)){Conflict = -1} #Dev build
        if((VMaj < EMaj) | (VMin < EMin) | (VRev < ERev)){Conflict = 1}  #Invalid version
        
        
        return Conflict
    }
    
    if(versionCheck(ConfigVer,MinConfigVer) == 1){printColor(vec(255,100,0),"[RSC] Config version outdated. Proceed with caution and consider updating.")}

}
